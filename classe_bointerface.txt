  METHOD m_check_obrigatorio_sales.

    DATA: lt_dados_cabecalho_brf  TYPE TABLE OF ty_8557_dados_cabecalho.
    DATA: lwa_dados_cabecalho_brf TYPE ty_8557_dados_cabecalho.

    DATA: lwa_filtro_mensagem TYPE ty_entrada_msg_brf_plus.
    DATA: l_auart_brf TYPE c LENGTH 4.

    DATA(lt_context) = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_bsark
                                                         value = wa_header_sales_order-customerpurchaseordertype ) ).

    NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
      EXPORTING
        i_t_context       = lt_context
        i_v_function_name = c_constante-c_f_8557_tipo_documento_sap
      IMPORTING
        e_value           = l_auart_brf ).

    " Se Tipo de Documento não for enviado na interface
    " e não tiver sido determinado no BRF+
    IF wa_header_sales_order-salesordertype IS INITIAL
    AND l_auart_brf IS INITIAL.

      lwa_filtro_mensagem = VALUE #( bsark = wa_header_sales_order-customerpurchaseordertype codigomessagem = c_constante-c_002 ).

      m_get_mensagem_brf_plus(
        EXPORTING
          i_wa_filtro_mensagem = lwa_filtro_mensagem
        IMPORTING
          e_wa_mensagem        = DATA(lwa_return_msg)
      ).

      IF lwa_return_msg IS NOT INITIAL.

        APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                        textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.

        " Se processo estiver cadastrado como Erro Critico
        " Abortar processamento imediatamente
        IF lwa_return_msg-errocritico IS NOT INITIAL.
          v_erro_critico = abap_true.
          RETURN.
        ENDIF.
      ENDIF.

    ELSE.

      SELECT SINGLE salesdocumenttype
        FROM i_salesdocumenttype
        WHERE salesdocumenttype = @wa_header_sales_order-salesordertype
        INTO @DATA(l_document_type).

      IF sy-subrc IS INITIAL.
        wa_header_sales_order_final-salesordertype = l_document_type.
      ELSE.
        " Se tipo de documento não existir no SAP tentar atribuir o documento
        " cadastrado na BRF+, caso também esteja Vazio, retornar mensagem de erro
        IF l_auart_brf IS INITIAL.

          lwa_filtro_mensagem = VALUE #( bsark = wa_header_sales_order-customerpurchaseordertype  codigomessagem = c_constante-c_002 ).

      wa_header_sales_order-customerpurchaseorderdate = wa_header_sales_order-requesteddeliverydate.

    " PASSO 2: Validação de pagamento antecipado
    "-------------------------------------------------------
    IF wa_brf_regras_negocio-validapagamentoantecipado IS NOT INITIAL.

      " Caso não seja informado o campo CUSTOMERPAYMENTTERMS retornar erro
      IF line_exists( t_itens_sales_order[ customerpaymentterms = space ] ).

        CLEAR: lwa_return_msg, lwa_filtro_mensagem.

        lwa_filtro_mensagem = VALUE #( bsark = wa_header_sales_order-customerpurchaseordertype  codigomessagem = c_constante-c_016 ).

        m_get_mensagem_brf_plus(
          EXPORTING
            i_wa_filtro_mensagem = lwa_filtro_mensagem
          IMPORTING
            e_wa_mensagem        = lwa_return_msg
        ).

        IF lwa_return_msg IS NOT INITIAL.

          APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                          textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.

          " Se processo estiver cadastrado como Erro Critico
          " Abortar processamento imediatamente
          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            RETURN.
          ENDIF.
        ENDIF.
      ELSE.

        DATA: l_pagamento_antecipado_brf TYPE ty_8557_pagamento_antec.

        LOOP AT t_itens_sales_order INTO DATA(lwa_itens).

          CLEAR lt_context.
          lt_context = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_bsark
                                                         value = wa_header_sales_order-customerpurchaseordertype )
                                                       ( name  = c_constante-c_dzterm
                                                         value = lwa_itens-customerpaymentterms ) ).

          NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
            EXPORTING
              i_t_context       = lt_context
              i_v_function_name = c_constante-c_f_8557_pagamento_antecipado
            IMPORTING
              e_value           = l_pagamento_antecipado_brf ).

          IF l_pagamento_antecipado_brf IS NOT INITIAL.
            wa_header_sales_order_final-salesordertype = l_pagamento_antecipado_brf.
            EXIT.
          ENDIF.
        ENDLOOP.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 3: Validação de quantidade drop
    " Aplica-se a TODOS os itens recebidos via t_itens_sales_order:
    " tanto novos (salesorderitem = space) quanto existentes
    " A regra é a mesma — resto da divisão deve ser zero
    "-------------------------------------------------------
    DATA(l_qtd_drop) = CONV i_salesdocumentitem-requestedquantity( wa_brf_regras_negocio-quantidadedrop ).

    IF l_qtd_drop IS NOT INITIAL.

      LOOP AT t_itens_sales_order INTO DATA(lwa_item_sales).

        DATA(l_resto) = CONV i_salesdocumentitem-requestedquantity(
          lwa_item_sales-requestedquantity MOD l_qtd_drop ).

        IF l_resto <> 0.
          DATA(l_erro_qtd_drop) = abap_true.
          EXIT.
        ENDIF.
      ENDLOOP.

      IF l_erro_qtd_drop IS NOT INITIAL.

        CLEAR: lwa_return_msg, lwa_filtro_mensagem.

        lwa_filtro_mensagem = VALUE #( bsark = wa_header_sales_order-customerpurchaseordertype  codigomessagem = c_constante-c_004 ).

        m_get_mensagem_brf_plus(
          EXPORTING
            i_wa_filtro_mensagem = lwa_filtro_mensagem
          IMPORTING
            e_wa_mensagem        = lwa_return_msg
        ).

        IF lwa_return_msg IS NOT INITIAL.

          APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                          textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.

          " Se processo estiver cadastrado como Erro Critico
          " Abortar processamento imediatamente
          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            EXIT.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    IF t_header_text_sales_order IS NOT INITIAL.

      t_header_text_sales_ord_final = VALUE #( FOR lwa_text_header IN t_header_text_sales_order
                                             ( language   = sy-langu
                                               longtextid = wa_brf_dados_cabecalho-idtexto
                                               longtext   = lwa_text_header-longtext ) ).

    ENDIF.

    IF t_header_partner_sales_order IS NOT INITIAL.

      IF wa_brf_dados_cabecalho-localentrega IS NOT INITIAL
      AND  line_exists( t_header_partner_sales_order[ partnerfunction = c_constante-c_we ] ).

        APPEND VALUE #( partnerfunction = wa_brf_dados_cabecalho-localentrega
                        customer = VALUE #( t_header_partner_sales_order[ 1 ]-customer OPTIONAL )
                      ) TO t_header_partner_sales_final.
      ENDIF.

      t_header_partner_sales_final = VALUE #( BASE t_header_partner_sales_final
                                              FOR lwa_partner_header IN t_header_partner_sales_order
                                            ( partnerfunction = lwa_partner_header-partnerfunction
                                              customer        = lwa_partner_header-customer ) ).

    ELSE.
      IF  wa_brf_dados_cabecalho-localentrega IS NOT INITIAL
      AND wa_header_sales_order-soldtoparty IS NOT INITIAL.
        APPEND VALUE #( partnerfunction = wa_brf_dados_cabecalho-localentrega
                        customer = wa_header_sales_order-soldtoparty ) TO t_header_partner_sales_final.
      ENDIF.
    ENDIF.

    DATA(l_plant) = VALUE #( t_itens_sales_order[ 1 ]-productionplant OPTIONAL ).

    IF l_plant IS NOT INITIAL.
      SELECT SINGLE plantcustomer
        FROM i_plant
        WHERE plant = @l_plant
        INTO @DATA(l_customer).

      IF sy-subrc IS INITIAL AND wa_brf_dados_cabecalho-localretirada IS NOT INITIAL.
        APPEND VALUE #( partnerfunction = wa_brf_dados_cabecalho-localretirada
                        customer = l_customer ) TO t_header_partner_sales_final.
      ENDIF.
    ENDIF.

    DATA: lt_dados_item_brf TYPE TABLE OF ty_8557_dados_item.
    CLEAR lt_context.
    lt_context = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_auart
                                                   value = wa_header_sales_order_final-salesordertype )
                                                 ( name  = c_constante-c_bsark
                                                   value = wa_header_sales_order-customerpurchaseordertype )
                                                 ( name  = c_constante-c_tipooperacao
                                                   value = c_constante-c_salesordercreate ) ).

    NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
      EXPORTING
        i_t_context       = lt_context
        i_v_function_name = c_constante-c_f_8557_dados_item
      IMPORTING
        e_value           = wa_brf_dados_item ).

    IF wa_brf_dados_item-lifsk IS NOT INITIAL .
      v_delivery_block_item = wa_brf_dados_item-lifsk.
    ENDIF.

    IF wa_brf_dados_item-modotransporte IS NOT INITIAL.
      " Atributo utilizado no metodo ZGLIF_USEREXIT_MOVE_FIELD_VBAP~M_SET_USEREXIT_MOVE_FIELD_VBAP
      " da classe ZSDCL_UPDATE_FIELDS_OIL_VBAP para determinar o valor de VBAP-OIC_MOT
      zsdcl_update_fields_oil_vbap=>v_modo_transporte = wa_brf_dados_item-modotransporte.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 4: Cálculo de data de faturamento ou
    "          validação de data do pedido
    "-------------------------------------------------------
    IF wa_brf_regras_negocio-calculadatafaturamento IS NOT INITIAL.
      m_calculo_data_faturamento_so( i_v_centro = l_plant ).
      IF v_erro_critico IS NOT INITIAL.
        RETURN.
      ENDIF.
    ELSE.
      TRY.
          DATA(l_zone) = cl_abap_context_info=>get_user_time_zone( ).
        CATCH cx_abap_context_info_error.
      ENDTRY.
      GET TIME STAMP FIELD DATA(l_timestamp).
      CONVERT TIME STAMP l_timestamp TIME ZONE l_zone INTO DATE DATA(l_data_atual).

      " Converte UTCLONG para data e hora
      CONVERT UTCLONG wa_header_sales_order-customerpurchaseorderdate
        TIME ZONE c_constante-c_utc
        INTO DATE DATA(l_customerpurchaseorderdate).

      IF l_customerpurchaseorderdate < l_data_atual.

        lwa_filtro_mensagem = VALUE #( bsark = wa_header_sales_order-customerpurchaseordertype  codigomessagem = c_constante-c_008 ).

        m_get_mensagem_brf_plus(
          EXPORTING
            i_wa_filtro_mensagem = lwa_filtro_mensagem
          IMPORTING
            e_wa_mensagem        = lwa_return_msg
        ).

        IF lwa_return_msg IS NOT INITIAL.

          APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                          textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.

          " Se processo estiver cadastrado como Erro Critico
          " Abortar processamento imediatamente
          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            RETURN.
          ENDIF.
        ENDIF.
      ELSE.
        v_request_delivery_date = l_customerpurchaseorderdate.

        IF wa_brf_regras_negocio-validahorarioentradapedido IS NOT INITIAL.

          " Data Atual + 1 Dia
          DATA(l_dia_seguinte) = CONV d( l_data_atual + 1 ).

          DATA: l_dia_semana_num TYPE cl_scal_utils=>scalv_weekday_number,
                l_hora_limite    TYPE syst_uzeit,
                lwa_timepoint    TYPE fdt_s_timepoint.

          IF l_customerpurchaseorderdate = l_dia_seguinte.

            TRY.
                l_zone = cl_abap_context_info=>get_user_time_zone( ).
              CATCH cx_abap_context_info_error.
            ENDTRY.

            GET TIME STAMP FIELD l_timestamp.
            CONVERT TIME STAMP l_timestamp TIME ZONE l_zone INTO DATE DATA(l_data_processamento).

            CALL METHOD cl_scal_utils=>date_compute_day
              EXPORTING
*               iv_date           = l_customerpurchaseorderdate
                iv_date           = l_data_processamento
              IMPORTING
                ev_weekday_number = l_dia_semana_num.

            DATA(l_dia_processamento) = COND #( WHEN l_dia_semana_num = 1 THEN c_constante-c_segunda
                                                WHEN l_dia_semana_num = 2 THEN c_constante-c_terca
                                                WHEN l_dia_semana_num = 3 THEN c_constante-c_quarta
                                                WHEN l_dia_semana_num = 4 THEN c_constante-c_quinta
                                                WHEN l_dia_semana_num = 5 THEN c_constante-c_sexta
                                                WHEN l_dia_semana_num = 6 THEN c_constante-c_sabado
                                                WHEN l_dia_semana_num = 7 THEN c_constante-c_domingo ).

            CLEAR lt_context.
            lt_context = VALUE zglcl_brf=>ty_tb_context( ( name = c_constante-c_diasemana value = l_dia_processamento )
                                                         ( name = c_constante-c_werks_ext value = l_plant ) ).

            NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
              EXPORTING
                i_t_context       = lt_context
                i_v_function_name = c_constante-c_f_8557_horario_limite_pedido
              IMPORTING
                e_value           = lwa_timepoint ).

            l_hora_limite = lwa_timepoint-time.

            GET TIME STAMP FIELD l_timestamp.
            CONVERT TIME STAMP l_timestamp TIME ZONE l_zone INTO TIME DATA(l_hora_processamento).

            IF l_hora_processamento > l_hora_limite.
              wa_header_sales_order_final-deliveryblockreason = wa_brf_regras_negocio-validahorarioentradapedido.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.

      "-----------------------------------------------------
      " Validação de operação base
      "-----------------------------------------------------
      IF wa_brf_regras_negocio-validaoperacaobase IS NOT INITIAL.

        SELECT centro,
               dataop AS dataoperacao,
               horainicio,
               horafim
          FROM zsd_ddl_i_centro_oper
          WHERE centro = @l_plant
            AND dataop <= @l_customerpurchaseorderdate
            ORDER BY dataop DESCENDING
          INTO @DATA(lwa_operacao_base)
          UP TO 1 ROWS.
        ENDSELECT.

        IF sy-subrc IS INITIAL.
          IF lwa_operacao_base-dataoperacao <> l_customerpurchaseorderdate.
            " A base não opera na data originalmente calculada.
            " Atualiza para a data anterior válida mais próxima.
            v_request_delivery_date = lwa_operacao_base-dataoperacao.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 5: Validação de calendarização do cliente
    "-------------------------------------------------------
    IF wa_brf_regras_negocio-validacalendarizacaocliente IS NOT INITIAL.

      m_valida_calendario_cliente(
        i_v_cliente = wa_header_sales_order-soldtoparty " Nº cliente
        i_v_data    = v_request_delivery_date           " Data
      ).

      IF v_erro_critico IS NOT INITIAL.
        RETURN.
      ENDIF.

    ENDIF.

    IF wa_header_sales_order-pricingdate IS NOT INITIAL.
      CONVERT UTCLONG wa_header_sales_order-pricingdate
        TIME ZONE c_constante-c_utc
        INTO DATE DATA(l_pricingdate).

      wa_header_sales_order_final-pricingdate = l_pricingdate.
    ELSE.
      wa_header_sales_order_final-pricingdate = v_request_delivery_date.
    ENDIF.


    m_check_ordem_ref_contrato( ).

  ENDMETHOD.

  METHOD m_mapping_bo_sales_ord_create.
**********************************************************************
*                                                                    *
*            ********************************************            *
*            *                IPIRANGA.                 *            *
*            *        CONFIDENCIAL E PROPRIETÁRIO       *            *
*            *       TODOS OS DIREITOS RESERVADOS       *            *
*            ********************************************            *
*                                                                    *
**********************************************************************
* NOME DO PROGRAMA    : ZSDCL_INT_ORQUESTRAD_SALES_INB               *
* TÍTULO DO PROGRAMA  : M_MAPPING_BO_SALES_ORD_CREATE                *
* DESCRIÇÃO           : Metodo mapeamento BO criação da Sales Order  *
* PROGRAMADOR         : EOLIVEIRA                                    *
* DATA                : 18/04/2026                                   *
* REQUEST             : DS4K902044                                   *
* RICEF               : 8557_GAP                                     *
*--------------------------------------------------------------------*
* HISTÓRICO DE MUDANÇAS:                                             *
* MUD  DATA        AUTOR          REQUEST #    DESCRIÇÃO             *
* 001  18/04/2026  EOLIVEIRA      DS4K902044   CÓDIGO INICIAL        *
**********************************************************************
    DATA: lwa_mapped   TYPE RESPONSE FOR MAPPED i_salesordertp,
          lwa_failed   TYPE RESPONSE FOR FAILED i_salesordertp,
          lwa_reported TYPE RESPONSE FOR REPORTED i_salesordertp.

    DATA l_cid_item TYPE string.

    MODIFY ENTITIES OF i_salesordertp

    "-------------------------------------------------------
    " HEADER
    "-------------------------------------------------------
      ENTITY salesorder
        CREATE AUTO FILL CID
        FIELDS (  salesordertype
                  salesorganization
                  distributionchannel
                  organizationdivision
                  soldtoparty
                  purchaseorderbycustomer
                  requesteddeliverydate
                  customerpurchaseorderdate
                  customerpurchaseordertype
                  deliveryblockreason
                  pricingdate
                  sddocumentreason
                  zz1_ultimosistemamodif_sdh
                  zz1_solicitante_sdh
                   )

        WITH VALUE #(
          (   %cid     = c_constante-c_h001
              %data = VALUE #(
              salesordertype          = wa_header_sales_order_final-salesordertype
              salesorganization       = wa_header_sales_order_final-salesorganization
              distributionchannel     = wa_header_sales_order_final-distributionchannel
              organizationdivision    = wa_header_sales_order_final-organizationdivision
              soldtoparty             = wa_header_sales_order_final-soldtoparty
              purchaseorderbycustomer = wa_header_sales_order_final-purchaseorderbycustomer
              requesteddeliverydate   = v_request_delivery_date
              customerpurchaseorderdate = v_customerpurchaseorderdate
              customerpurchaseordertype = wa_header_sales_order_final-customerpurchaseordertype
              deliveryblockreason     = wa_header_sales_order_final-deliveryblockreason
              pricingdate             = wa_header_sales_order_final-pricingdate
              sddocumentreason        = wa_header_sales_order_final-sddocumentreason
              zz1_ultimosistemamodif_sdh = wa_header_sales_order_final-zz1_ultimosistemamodif_sdh
              zz1_solicitante_sdh = wa_header_sales_order_final-zz1_solicitante_sdh
               ) )
        )

    "-------------------------------------------------------
    " HEADER PARTNER
    "-------------------------------------------------------
      CREATE BY \_partner AUTO FILL CID
      FIELDS ( partnerfunctionforedit
               customer )

      WITH VALUE #(
        FOR lwa_partner IN t_header_partner_sales_final
         (  %cid_ref = c_constante-c_h001
          %target = VALUE #(
            ( partnerfunction        = lwa_partner-partnerfunction
              partnerfunctionforedit = lwa_partner-partnerfunction

  METHOD m_calcular_lead_time.
**********************************************************************
*                                                                    *
*            ********************************************            *
*            *                IPIRANGA.                 *            *
*            *        CONFIDENCIAL E PROPRIETÁRIO       *            *
*            *       TODOS OS DIREITOS RESERVADOS       *            *
*            ********************************************            *
*                                                                    *
**********************************************************************
* NOME DO PROGRAMA    : ZSDCL_INT_ORQUESTRAD_SALES_INB               *
* TÍTULO DO PROGRAMA  : M_CALCULAR_LEAD_TIME                         *
* DESCRIÇÃO           : Metodo calcular lead time em horas           *
* PROGRAMADOR         : EOLIVEIRA                                    *
* DATA                : 18/04/2026                                   *
* REQUEST             : DS4K902044                                   *
* RICEF               : 8557_GAP                                     *
*--------------------------------------------------------------------*
* HISTÓRICO DE MUDANÇAS:                                             *
* MUD  DATA        AUTOR          REQUEST #    DESCRIÇÃO             *
* 001  18/04/2026  EOLIVEIRA      DS4K902044   CÓDIGO INICIAL        *
**********************************************************************
    DATA: l_timestamp TYPE timestampl,
          l_result_ts TYPE timestampl,
          l_date      TYPE d,
          l_time      TYPE t,
          l_hours     TYPE i,
          l_seconds   TYPE i.

    IF i_v_data IS INITIAL.
      RETURN.
    ENDIF.

    " Converte UTCLONG para data e hora
    CONVERT UTCLONG i_v_data
      TIME ZONE c_constante-c_utc
      INTO DATE l_date
           TIME l_time.

    " Converte data + hora para TIMESTAMPL
    CONVERT DATE l_date
            TIME l_time
            INTO TIME STAMP l_timestamp
            TIME ZONE c_constante-c_utc.

    " Lead time em horas
    l_hours = i_v_leadtime.

    " Caso não exista lead time, retorna apenas a data original
    IF l_hours IS INITIAL.
      r_v_resultado = l_date.
      RETURN.
    ENDIF.

    " Converte horas para segundos
    l_seconds = l_hours * 3600.

    " Subtrai as horas
    cl_abap_tstmp=>subtractsecs(
      EXPORTING
        tstmp   = l_timestamp
        secs    = l_seconds
      RECEIVING
        r_tstmp = l_result_ts
    ).

    " Extrai apenas a data final
    CONVERT TIME STAMP l_result_ts
      TIME ZONE c_constante-c_utc
      INTO DATE r_v_resultado
           TIME l_time.

  ENDMETHOD.

                                                                                               plant .

      LOOP AT lt_contratos_validos INTO DATA(lwa_contrato_validos)

            EXIT.
          ELSE.
            CLEAR l_sem_ref_contrato.
          ENDIF.
        ENDIF.
      ENDLOOP.

      IF lt_grupo_contrato IS NOT INITIAL.

      ELSE.
        IF wa_brf_regras_contrato-contratoobrigatorio IS NOT INITIAL.

              longtext          = lwa_text-longtext ) )
        )
      )

      CREATE BY \_item AUTO FILL CID
      FIELDS ( product
               requestedquantity
               plant )

      WITH VALUE #(
        FOR lwa_item IN t_itens_contract (
          %cid_ref = c_constante-c_h001
          %target = VALUE #(
            ( %cid                  = lwa_item-cid
              product               = lwa_item-material
              requestedquantity     = CONV i_salescontractitemtp-requestedquantity( lwa_item-requestedquantity )
              plant                 = lwa_item-productionplant
              ) )
        )
      )

      ENTITY salescontractitem
        CREATE BY \_itemtext
        FIELDS ( languageforedit
                 longtextidforedit
                 longtext )


        WITH VALUE #(
          FOR lwa_item IN t_itens_contract
          FOR lwa_item_text IN t_itens_text_contract
          WHERE ( cid_ref = lwa_item-cid )
           ( %cid_ref = lwa_item-cid
            %target = VALUE #(
              ( %cid = lwa_item_text-cid
                languageforedit   = wa_brf_dados_item-idtexto
                longtextidforedit = wa_brf_dados_item-idtexto
                longtext          = lwa_item_text-longtext ) )
          )
        )

      ENTITY salescontractitem
        CREATE BY \_itempricingelement
        FIELDS ( conditiontype
                 conditionrateamount
                 conditioncurrency )

        WITH VALUE #(
          FOR lwa_item IN t_itens_contract
          FOR lwa_price IN t_itens_pricing_contract
          WHERE ( cid_ref = lwa_item-cid )
           ( %cid_ref = lwa_item-cid
            %target = VALUE #(
              ( %cid = lwa_price-cid
                conditioncurrency   = lwa_price-conditioncurrency " 'BRL'
                conditiontype       = COND #( WHEN wa_brf_dados_item-kschl IS NOT INITIAL
                                              THEN wa_brf_dados_item-kschl
                                              ELSE lwa_price-conditiontype )
                conditionrateamount =
                   CONV i_slscontritempricingelementtp-conditionrateamount( lwa_price-conditionratevalue ) ) )
          )
        )

      MAPPED   DATA(lwa_mapped)
      FAILED   DATA(lwa_failed)
      REPORTED DATA(lwa_reported).

    READ ENTITIES OF i_salescontracttp
     ENTITY salescontract
      FROM VALUE #( ( salescontract = space ) )
            RESULT   DATA(lt_result)
            REPORTED lwa_reported
            FAILED   DATA(lwa_failed_read).

    wa_mapped_sales_contract = lwa_mapped.

    "-------------------------------------------------------

    IF lwa_failed-salescontract                IS NOT INITIAL OR
       lwa_failed-salescontractitem            IS NOT INITIAL OR
       lwa_failed-salescontractitemtext        IS NOT INITIAL OR
       lwa_failed-salescontracttext            IS NOT INITIAL OR
       lwa_failed-slscontractitempricingelement   IS NOT INITIAL OR
       lwa_failed-salescontractpricingelement  IS NOT INITIAL.

      v_erro_critico = abap_true.

      " Header
      LOOP AT lwa_reported-salescontract INTO DATA(lwa_rep_sc).
        CHECK lwa_rep_sc-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_sc-%msg ).
      ENDLOOP.

      " Item
      LOOP AT lwa_reported-salescontractitem INTO DATA(lwa_rep_item).
        CHECK lwa_rep_item-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item-%msg ).
      ENDLOOP.

      " Item Text
      LOOP AT lwa_reported-salescontractitemtext INTO DATA(lwa_rep_item_text).
        CHECK lwa_rep_item_text-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item_text-%msg ).
      ENDLOOP.

      " Header Text
      LOOP AT lwa_reported-salescontracttext INTO DATA(lwa_rep_text).
        CHECK lwa_rep_text-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_text-%msg ).
      ENDLOOP.

      " Item Pricing Element
      LOOP AT lwa_reported-slscontractitempricingelement INTO DATA(lwa_rep_item_price).
        CHECK lwa_rep_item_price-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item_price-%msg ).
      ENDLOOP.

      " Header Pricing Element
      LOOP AT lwa_reported-salescontractpricingelement INTO DATA(lwa_rep_price).
        CHECK lwa_rep_price-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_price-%msg ).
      ENDLOOP.

    ENDIF.

  ENDMETHOD.


  METHOD m_check_sales_change.
**********************************************************************
*                                                                    *
*            ********************************************            *
*            *                IPIRANGA.                 *            *
*            *        CONFIDENCIAL E PROPRIETÁRIO       *            *
*            *       TODOS OS DIREITOS RESERVADOS       *            *
*            ********************************************            *
*                                                                    *
**********************************************************************
* NOME DO PROGRAMA    : ZSDCL_INT_ORQUESTRAD_SALES_INB               *
* TÍTULO DO PROGRAMA  : M_CHECK_SALES_CHANGE                         *
* DESCRIÇÃO           : Metodo validar regras de alteração SO        *
* PROGRAMADOR         : EOLIVEIRA                                    *
* DATA                : 18/04/2026                                   *
* REQUEST             : DS4K902044                                   *
* RICEF               : 8557_GAP                                     *
*--------------------------------------------------------------------*
* HISTÓRICO DE MUDANÇAS:                                             *
* MUD  DATA        AUTOR          REQUEST #    DESCRIÇÃO             *
* 001  18/04/2026  EOLIVEIRA      DS4K902044   CÓDIGO INICIAL        *
**********************************************************************
    DATA: lwa_filtro_mensagem TYPE ty_entrada_msg_brf_plus,
          lwa_return_msg      TYPE ty_retorno_msg_brf_plus.

    "-------------------------------------------------------
    " Separa itens existentes (alteração) de novos (criação)
    " Itens novos têm salesorderitem = space pois ainda
    " não possuem número de item na ordem SAP
    "-------------------------------------------------------
    DATA lt_itens_existentes TYPE ty_t_itens_sales_order.
    lt_itens_existentes = VALUE #( FOR lwa_item IN t_itens_sales_order
                                 WHERE ( salesorderitem <> space )
                                       ( lwa_item ) ).

    "-------------------------------------------------------
    " PASSO 1: Valida existência da ordem no SAP
    "-------------------------------------------------------
    SELECT SINGLE salesdocument,
                  salesdocumenttype,
                  soldtoparty,
                  purchaseorderbycustomer,
                  incotermsclassification,
                  incotermslocation1,
                  sddocumentreason,
                  pricingdate,
                  customerpurchaseorderdate,
                  customerpurchaseordertype
      FROM i_salesdocument
      WHERE salesdocument = @wa_header_sales_order-salesorder
      INTO @DATA(lwa_salesdocument).

    IF sy-subrc IS NOT INITIAL.

      lwa_filtro_mensagem = VALUE #(
        bsark          = lwa_salesdocument-customerpurchaseordertype
        codigomessagem = c_constante-c_013 ).

      m_get_mensagem_brf_plus(
        EXPORTING
          i_wa_filtro_mensagem = lwa_filtro_mensagem
        IMPORTING
          e_wa_mensagem        = lwa_return_msg ).

      IF lwa_return_msg IS NOT INITIAL.

        APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                        textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.

        IF lwa_return_msg-errocritico IS NOT INITIAL.
          v_erro_critico = abap_true.
          RETURN.
        ENDIF.
      ENDIF.

    ELSE.

      wa_header_sales_order_bd = CORRESPONDING #( lwa_salesdocument ).

      "-------------------------------------------------------
      " Popula header_final com dados da ordem existente
      " necessário para reuso nos métodos de validação BRF+
      " Aplica lógica de controle: usa valor novo apenas se
      " campo foi alterado E está habilitado no controle ('01')
      "-------------------------------------------------------
      wa_header_sales_order_final = VALUE #( BASE wa_header_sales_order_bd

        purchaseorderbycustomer   = COND #(
          WHEN wa_header_sales_order_bd-purchaseorderbycustomer
            <> wa_header_sales_order-purchaseorderbycustomer
          AND  wa_header_sales_control-purchaseorderbycustomer = c_constante-c_01
          THEN wa_header_sales_order-purchaseorderbycustomer
          ELSE wa_header_sales_order_bd-purchaseorderbycustomer )

        incotermsclassification   = COND #(
          WHEN wa_header_sales_order_bd-incotermsclassification
            <> wa_header_sales_order-incotermsclassification
          AND  wa_header_sales_control-incotermsclassification = c_constante-c_01
          THEN wa_header_sales_order-incotermsclassification
          ELSE wa_header_sales_order_bd-incotermsclassification )

        incotermslocation1        = COND #(
          WHEN wa_header_sales_order_bd-incotermslocation1
            <> wa_header_sales_order-incotermslocation1
          AND  wa_header_sales_control-incotermslocation1 = c_constante-c_01
          THEN wa_header_sales_order-incotermslocation1
          ELSE wa_header_sales_order_bd-incotermslocation1 )

        sddocumentreason          = COND #(
          WHEN wa_header_sales_order_bd-sddocumentreason
            <> wa_header_sales_order-sddocumentreason
          AND  wa_header_sales_control-sddocumentreason = c_constante-c_01
          THEN wa_header_sales_order-sddocumentreason
          ELSE wa_header_sales_order_bd-sddocumentreason )

        pricingdate               = COND #(
          WHEN wa_header_sales_order_bd-pricingdate
            <> m_convert_formato_data( i_v_data = wa_header_sales_order-pricingdate )
          AND  wa_header_sales_control-pricingdate = c_constante-c_01
          THEN m_convert_formato_data( i_v_data = wa_header_sales_order-pricingdate )
          ELSE wa_header_sales_order_bd-pricingdate )

        customerpurchaseorderdate = COND #(
          WHEN wa_header_sales_order_bd-customerpurchaseorderdate
            <> m_convert_formato_data( i_v_data = wa_header_sales_order-customerpurchaseorderdate )
          AND  wa_header_sales_control-customerpurchaseorderdate = c_constante-c_01
          THEN m_convert_formato_data( i_v_data = wa_header_sales_order-customerpurchaseorderdate )
          ELSE wa_header_sales_order_bd-customerpurchaseorderdate )

        customerpurchaseordertype = COND #(
          WHEN wa_header_sales_order_bd-customerpurchaseordertype
            <> wa_header_sales_order-customerpurchaseordertype
          AND  wa_header_sales_control-customerpurchaseordertype = c_constante-c_01
          THEN wa_header_sales_order-customerpurchaseordertype
          ELSE wa_header_sales_order_bd-customerpurchaseordertype )

        zz1_ultimosistemamodif_sdh = COND #(
          WHEN wa_header_sales_order_bd-zz1_ultimosistemamodif_sdh
            <> wa_header_sales_order-zz1_ultimosistemamodif_sdh
          AND  wa_header_sales_control-zz1_ultimosistemamodif_sdh = c_constante-c_01
          THEN wa_header_sales_order-zz1_ultimosistemamodif_sdh
          ELSE wa_header_sales_order_bd-zz1_ultimosistemamodif_sdh )

        zz1_solicitante_sdh       = COND #(
          WHEN wa_header_sales_order_bd-zz1_solicitante_sdh
            <> wa_header_sales_order-zz1_solicitante_sdh
          AND  wa_header_sales_control-zz1_solicitante_sdh = c_constante-c_01
          THEN wa_header_sales_order-zz1_solicitante_sdh
          ELSE wa_header_sales_order_bd-zz1_solicitante_sdh )
      ).

      DATA(lt_context) = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_auart
                                                           value = wa_header_sales_order_final-salesordertype )
                                                         ( name  = c_constante-c_bsark
                                                           value = wa_header_sales_order-customerpurchaseordertype )
                                                         ( name  = c_constante-c_tipooperacao
                                                           value = c_constante-c_salesorderchange ) ).

      NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
        EXPORTING
          i_t_context       = lt_context
          i_v_function_name = c_constante-c_f_8557_dados_cabecalho
        IMPORTING
          e_value           = wa_brf_dados_cabecalho ).

    ENDIF.

    "-------------------------------------------------------
    " PASSO 2: Valida alteração de tipo de documento
    "-------------------------------------------------------
    IF wa_header_sales_control-salesordertype =  c_constante-c_01
    AND wa_header_sales_order-salesordertype  <> lwa_salesdocument-salesdocumenttype.

      lwa_filtro_mensagem = VALUE #(
        bsark          = lwa_salesdocument-customerpurchaseordertype
        codigomessagem = c_constante-c_014 ).

      m_get_mensagem_brf_plus(
        EXPORTING
          i_wa_filtro_mensagem = lwa_filtro_mensagem
        IMPORTING
          e_wa_mensagem        = lwa_return_msg ).

      IF lwa_return_msg IS NOT INITIAL.

        APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                        textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.

        IF lwa_return_msg-errocritico IS NOT INITIAL.
          v_erro_critico = abap_true.
          RETURN.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 3: Lógica BOP apenas para itens existentes
    " Itens novos (salesorderitem = space) são ignorados
    " pois nunca estarão na tabela BOP
    "-------------------------------------------------------
    IF lt_itens_existentes IS NOT INITIAL.

      DATA lr_itens TYPE RANGE OF zsd_ddl_i_bop_log-salesdocumentitem.

      lr_itens = VALUE #( FOR lwa_item IN lt_itens_existentes
                        ( sign   = c_constante-c_i
                          option = c_constante-c_eq
                          low    = lwa_item-salesorderitem ) ).

      SELECT salesdocument, salesdocumentitem
        FROM zsd_ddl_i_bop_log
        WHERE salesdocument     = @wa_header_sales_order-salesorder
          AND salesdocumentitem IN @lr_itens
        INTO TABLE @DATA(lt_ordens_bop).

      IF lt_ordens_bop IS INITIAL.
        "----------------------------------------------------
        " Nenhum item existente passou pelo BOP
        " -> caminho 1 todos os campos podem ser atualizados
        "----------------------------------------------------
        v_tipo_mapping_change_ordem = c_constante-c_1.

        "-------------------------------------------------------
        " READ itens existentes para comparação de valores
        "-------------------------------------------------------
        READ ENTITIES OF i_salesordertp
          ENTITY salesorderitem
            FIELDS ( salesorderitem
                     product
                     requestedquantity
                     plant
                     pricingdate
                     customerpaymentterms
                     salesdocumentrjcnreason
                     zz1_ultimomodifitem_sdi
                     zz1_motivocancelamento_sdi
                     zz1_horainiciograde_sdi
                     zz1_horafimgrade_sdi )
          WITH VALUE #( FOR lwa_item IN t_itens_sales_order
                          WHERE ( salesorderitem <> space )
                      ( salesorder     = wa_header_sales_order-salesorder
                        salesorderitem = lwa_item-salesorderitem ) )
            RESULT   DATA(lt_itens_bd)
            FAILED   DATA(lwa_failed)
            REPORTED DATA(lwa_reported).

        t_itens_sales_order_final = VALUE #(  FOR lwa_itens IN t_itens_sales_order
                          WHERE ( salesorderitem <> space )
                                  FOR lwa_item_control IN t_itens_sales_control
                          WHERE ( cid = lwa_itens-cid )
                                  FOR lwa_itens_bd     IN lt_itens_bd
                          WHERE ( salesorderitem = lwa_itens-salesorderitem )
                     ( salesorderitem             = lwa_itens-salesorderitem
                       product                    =
                       COND #(
                        WHEN lwa_itens_bd-product <> lwa_itens-material
                        AND  lwa_item_control-material = c_constante-c_01
                        THEN lwa_itens-material
                        ELSE lwa_itens_bd-product )

                       requestedquantity   = COND #( WHEN CONV i_salesorderitemtp-requestedquantity( lwa_itens-requestedquantity )
                                                           <> lwa_itens_bd-requestedquantity
                                                         AND  lwa_item_control-requestedquantity = c_constante-c_01
                                                    THEN CONV i_salesorderitemtp-requestedquantity( lwa_itens-requestedquantity )
                                                    ELSE lwa_itens_bd-requestedquantity )

                      plant               = COND #(
                        WHEN lwa_itens-productionplant <> lwa_itens_bd-plant
                        AND  lwa_item_control-productionplant = c_constante-c_01
                        THEN lwa_itens-productionplant
                        ELSE lwa_itens_bd-plant )

                      pricingdate         = COND #(
                        WHEN m_convert_formato_data( i_v_data = lwa_itens-pricingdate ) <> lwa_itens_bd-pricingdate
                        AND  lwa_item_control-pricingdate = c_constante-c_01
                        THEN m_convert_formato_data( i_v_data = lwa_itens-pricingdate )
                        ELSE lwa_itens_bd-pricingdate )

                      customerpaymentterms = COND #(
                        WHEN lwa_itens-customerpaymentterms <> lwa_itens_bd-customerpaymentterms
                        AND  lwa_item_control-customerpaymentterms = c_constante-c_01
                        THEN lwa_itens-customerpaymentterms
                        ELSE lwa_itens_bd-customerpaymentterms )

                      salesdocumentrjcnreason = COND #(
                        WHEN lwa_itens-salesdocumentrjcnreason <> lwa_itens_bd-salesdocumentrjcnreason
                        AND  lwa_item_control-salesdocumentrjcnreason = c_constante-c_01
                        THEN lwa_itens-salesdocumentrjcnreason
                        ELSE lwa_itens_bd-salesdocumentrjcnreason )

                      zz1_ultimomodifitem_sdi = COND #(
                        WHEN lwa_itens-zz1_ultimomodifitem_sdi <> lwa_itens_bd-zz1_ultimomodifitem_sdi
                        AND  lwa_item_control-zz1_ultimomodifitem_sdi = c_constante-c_01
                        THEN lwa_itens-zz1_ultimomodifitem_sdi
                        ELSE lwa_itens_bd-zz1_ultimomodifitem_sdi )

                      zz1_motivocancelamento_sdi = COND #(
                        WHEN lwa_itens-zz1_motivocancelamento_sdi <> lwa_itens_bd-zz1_motivocancelamento_sdi
                        AND  lwa_item_control-zz1_motivocancelamento_sdi = c_constante-c_01
                        THEN lwa_itens-zz1_motivocancelamento_sdi
                        ELSE lwa_itens_bd-zz1_motivocancelamento_sdi )

                      zz1_horainiciograde_sdi = COND #(
                        WHEN lwa_itens-zz1_horainiciograde_sdi <> lwa_itens_bd-zz1_horainiciograde_sdi
                        AND  lwa_item_control-zz1_horainiciograde_sdi = c_constante-c_01
                        THEN lwa_itens-zz1_horainiciograde_sdi
                        ELSE lwa_itens_bd-zz1_horainiciograde_sdi )

                      zz1_horafimgrade_sdi = COND #(
                        WHEN lwa_itens-zz1_horafimgrade_sdi <> lwa_itens_bd-zz1_horafimgrade_sdi
                        AND  lwa_item_control-zz1_horafimgrade_sdi = c_constante-c_01
                        THEN lwa_itens-zz1_horafimgrade_sdi
                        ELSE lwa_itens_bd-zz1_horafimgrade_sdi )
                                     ) ).

        CLEAR lt_context.
        lt_context = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_auart
                                                       value = lwa_salesdocument-salesdocumenttype )
                                                     ( name  = c_constante-c_bsark
                                                       value = lwa_salesdocument-customerpurchaseordertype )
                                                     ( name  = c_constante-c_tipooperacao
                                                       value = c_constante-c_salesorderchange ) ).

        NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
          EXPORTING
            i_t_context       = lt_context
            i_v_function_name = c_constante-c_f_8557_dados_item
          IMPORTING
            e_value           = wa_brf_dados_item ).

      ELSE.
        "----------------------------------------------------
        " Ao menos um item passou pelo BOP
        " -> verifica status de faturamento por item
        "----------------------------------------------------
        DATA(l_algum_faturado)  = abap_false.

        SELECT a~referencesddocument,
               a~referencesddocumentitem,
               a~deliveryrelatedbillingstatus
          FROM i_deliverydocumentitem AS a
          INNER JOIN @lt_ordens_bop AS b
          ON  a~referencesddocument     = b~salesdocument
          AND a~referencesddocumentitem = b~salesdocumentitem
          WHERE deliveryrelatedbillingstatus = @c_constante-c_c
          INTO TABLE @DATA(lt_billing_status).

        IF sy-subrc IS INITIAL.
          SORT lt_billing_status BY referencesddocument referencesddocumentitem.
        ENDIF.

        LOOP AT lt_itens_existentes INTO DATA(lwa_item_check).

          READ TABLE lt_billing_status INTO DATA(lwa_billing)
                                       WITH KEY referencesddocument = wa_header_sales_order-salesorder
                                                referencesddocumentitem = lwa_item_check-salesorderitem
                                                BINARY SEARCH.

          IF sy-subrc IS NOT INITIAL.
            " Item sem entrega associada -> ainda não faturado
            CONTINUE.
          ELSE.
            l_algum_faturado  = abap_true.
          ENDIF.

        ENDLOOP.

        IF l_algum_faturado = abap_true.
          "----------------------------------------------------
          " Algum itens existentes esta faturados
          " -> bloqueia qualquer alteração
          "----------------------------------------------------
          lwa_filtro_mensagem = VALUE #(
            bsark          = lwa_salesdocument-customerpurchaseordertype
            codigomessagem = c_constante-c_015 ).

          m_get_mensagem_brf_plus(
            EXPORTING
              i_wa_filtro_mensagem = lwa_filtro_mensagem
            IMPORTING
              e_wa_mensagem        = lwa_return_msg ).

          IF lwa_return_msg IS NOT INITIAL.
            APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                            textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.
          ENDIF.

          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            RETURN.
          ENDIF.
        ELSE.
          "----------------------------------------------------
          " Pós-BOP mas ainda não totalmente faturado
          " -> caminho 2 (campos restritos pode ser atualizados)
          "----------------------------------------------------
          v_tipo_mapping_change_ordem = c_constante-c_2.
        ENDIF.

          FOR lwa_item IN t_itens_sales_order  WHERE ( salesorderitem <> space )

          FOR lwa_itens_bd IN lt_itens_bd WHERE ( salesorderitem = lwa_item-salesorderitem )

            %key-salesorderitem     = lwa_item-salesorderitem

                                              ELSE lwa_itens_bd-zz1_ultimomodifitem_sdi ) ) )

       ENTITY salesorderitemtext
                UPDATE
                  FIELDS ( longtext )
                  WITH VALUE #( FOR lwa_item IN t_itens_sales_order
                                WHERE ( salesorderitem <> space )
                                FOR lwa_item_text IN t_itens_text_sales_order
                                WHERE ( cid_ref = lwa_item-cid )

                  ( %key    = VALUE #( salesorder      = wa_header_sales_order-salesorder
                                       salesorderitem  = lwa_item-salesorderitem
                                       language        = lwa_text_bd-language
                                       longtextid      = lwa_text_bd-longtextid )

     MAPPED   DATA(lwa_mapped)

  METHOD m_check_regras_negocio_brf.
**********************************************************************
*                                                                    *
*            ********************************************            *
*            *                IPIRANGA.                 *            *
*            *        CONFIDENCIAL E PROPRIETÁRIO       *            *
*            *       TODOS OS DIREITOS RESERVADOS       *            *
*            ********************************************            *
*                                                                    *
**********************************************************************
* NOME DO PROGRAMA    : ZSDCL_INT_ORQUESTRAD_SALES_INB               *
* TÍTULO DO PROGRAMA  : M_CHECK_REGRAS_NEGOCIO_BRF                   *
* DESCRIÇÃO           : Metodo validar regras de negocio via BRF+    *
* PROGRAMADOR         : EOLIVEIRA                                    *
* DATA                : 18/04/2026                                   *
* REQUEST             : DS4K902044                                   *
* RICEF               : 8557_GAP                                     *
*--------------------------------------------------------------------*
* HISTÓRICO DE MUDANÇAS:                                             *
* MUD  DATA        AUTOR          REQUEST #    DESCRIÇÃO             *
* 001  18/04/2026  EOLIVEIRA      DS4K902044   CÓDIGO INICIAL        *
**********************************************************************
    DATA: lwa_filtro_mensagem TYPE ty_entrada_msg_brf_plus,
          lwa_return_msg      TYPE ty_retorno_msg_brf_plus.

    DATA(lt_context) = VALUE zglcl_brf=>ty_tb_context(
      ( name = c_constante-c_auart        value = wa_header_sales_order_final-salesordertype )
      ( name = c_constante-c_bsark        value = wa_header_sales_order_final-customerpurchaseordertype )
      ( name = c_constante-c_inco1        value = wa_header_sales_order_final-incotermsclassification )
      ( name = c_constante-c_tipooperacao value = c_constante-c_salesorderitemcreate ) ).

    NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
      EXPORTING
        i_t_context       = lt_context
        i_v_function_name = c_constante-c_f_8557_regras_negocio
      IMPORTING
        e_value           = wa_brf_regras_negocio ).

    "-------------------------------------------------------
    " PASSO 2: Validação de pagamento antecipado
    "-------------------------------------------------------
    IF wa_brf_regras_negocio-validapagamentoantecipado IS NOT INITIAL.

      DATA(lwa_primeiro_item) = VALUE #( t_itens_sales_order[ 1 ] OPTIONAL ).

      SELECT SINGLE paymentterms
        FROM zsd_ddl_i_plantpaymentterms
        WHERE plant    = @lwa_primeiro_item-productionplant
          AND customer = @wa_header_sales_order_final-soldtoparty
        INTO @DATA(l_zterm).

      IF sy-subrc IS INITIAL.

        DATA: l_pagamento_antecipado_brf TYPE ty_8557_pagamento_antec.

        CLEAR lt_context.
        lt_context = VALUE zglcl_brf=>ty_tb_context(
          ( name = c_constante-c_bsark  value = wa_header_sales_order_final-customerpurchaseordertype )
          ( name = c_constante-c_dzterm value = l_zterm ) ).

        NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
          EXPORTING
            i_t_context       = lt_context
            i_v_function_name = c_constante-c_f_8557_pagamento_antecipado
          IMPORTING
            e_value           = l_pagamento_antecipado_brf ).

        IF l_pagamento_antecipado_brf IS NOT INITIAL.
          wa_header_sales_order_final-salesordertype = l_pagamento_antecipado_brf.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 3: Validação de quantidade drop
    " Aplica-se a TODOS os itens recebidos via t_itens_sales_order:
    " tanto novos (salesorderitem = space) quanto existentes
    " A regra é a mesma — resto da divisão deve ser zero
    "-------------------------------------------------------
    DATA(l_qtd_drop) = CONV i_salesdocumentitem-requestedquantity(
      wa_brf_regras_negocio-quantidadedrop ).

    IF l_qtd_drop IS NOT INITIAL.

      LOOP AT t_itens_sales_order INTO DATA(lwa_item_drop).

        DATA(l_resto) = CONV i_salesdocumentitem-requestedquantity(
          lwa_item_drop-requestedquantity MOD l_qtd_drop ).

        IF l_resto <> 0.
          DATA(l_erro_qtd_drop) = abap_true.
          EXIT.
        ENDIF.
      ENDLOOP.

      IF l_erro_qtd_drop IS NOT INITIAL.

        lwa_filtro_mensagem = VALUE #(
          bsark          = wa_header_sales_order_final-customerpurchaseordertype
          codigomessagem = c_constante-c_004 ).

        m_get_mensagem_brf_plus(
          EXPORTING
            i_wa_filtro_mensagem = lwa_filtro_mensagem
          IMPORTING
            e_wa_mensagem        = lwa_return_msg ).

        IF lwa_return_msg IS NOT INITIAL.
          APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                          textomensagem  = lwa_return_msg-textomensagem )
            TO t_log_message_error.
          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            RETURN.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 4: Cálculo de data de faturamento ou
    "          validação de data do pedido
    "-------------------------------------------------------
    DATA(l_plant) = VALUE #( t_itens_sales_order[ 1 ]-productionplant OPTIONAL ).

    IF wa_brf_regras_negocio-calculadatafaturamento IS NOT INITIAL.

      m_calculo_data_faturamento_so( i_v_centro = l_plant ).

      IF v_erro_critico IS NOT INITIAL.
        RETURN.
      ENDIF.

    ELSE.

      TRY.
          DATA(l_zone) = cl_abap_context_info=>get_user_time_zone( ).
        CATCH cx_abap_context_info_error.
      ENDTRY.

      GET TIME STAMP FIELD DATA(l_timestamp).
      CONVERT TIME STAMP l_timestamp TIME ZONE l_zone
        INTO DATE DATA(l_data_atual).

      IF wa_header_sales_order_final-customerpurchaseorderdate < l_data_atual.

        lwa_filtro_mensagem = VALUE #(
          bsark          = wa_header_sales_order_final-customerpurchaseordertype
          codigomessagem = c_constante-c_008 ).

        m_get_mensagem_brf_plus(
          EXPORTING
            i_wa_filtro_mensagem = lwa_filtro_mensagem
          IMPORTING
            e_wa_mensagem        = lwa_return_msg ).

        IF lwa_return_msg IS NOT INITIAL.
          APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                          textomensagem  = lwa_return_msg-textomensagem )
            TO t_log_message_error.
          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            RETURN.
          ENDIF.
        ENDIF.

      ELSE.

        v_request_delivery_date = wa_header_sales_order_final-customerpurchaseorderdate.

        IF wa_brf_regras_negocio-validahorarioentradapedido IS NOT INITIAL.

          DATA(l_dia_seguinte) = CONV d( l_data_atual + 1 ).

          IF wa_header_sales_order_final-customerpurchaseorderdate = l_dia_seguinte.

            DATA: l_dia_semana_num TYPE cl_scal_utils=>scalv_weekday_number,
                  l_hora_limite    TYPE syst_uzeit,
                  lwa_timepoint    TYPE fdt_s_timepoint.

            TRY.
                l_zone = cl_abap_context_info=>get_user_time_zone( ).
              CATCH cx_abap_context_info_error.
            ENDTRY.

            GET TIME STAMP FIELD l_timestamp.
            CONVERT TIME STAMP l_timestamp TIME ZONE l_zone INTO DATE DATA(l_data_processamento).

            CALL METHOD cl_scal_utils=>date_compute_day
              EXPORTING
*               iv_date           = wa_header_sales_order_final-customerpurchaseorderdate
                iv_date           = l_data_processamento
              IMPORTING
                ev_weekday_number = l_dia_semana_num.

            DATA(l_dia_processamento) = COND #(
              WHEN l_dia_semana_num = 1 THEN c_constante-c_segunda
              WHEN l_dia_semana_num = 2 THEN c_constante-c_terca
              WHEN l_dia_semana_num = 3 THEN c_constante-c_quarta
              WHEN l_dia_semana_num = 4 THEN c_constante-c_quinta
              WHEN l_dia_semana_num = 5 THEN c_constante-c_sexta
              WHEN l_dia_semana_num = 6 THEN c_constante-c_sabado
              WHEN l_dia_semana_num = 7 THEN c_constante-c_domingo ).

            CLEAR lt_context.
            lt_context = VALUE zglcl_brf=>ty_tb_context(
              ( name = c_constante-c_diasemana value = l_dia_processamento )
              ( name = c_constante-c_werks_ext value = l_plant ) ).

            NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
              EXPORTING
                i_t_context       = lt_context
                i_v_function_name = c_constante-c_f_8557_horario_limite_pedido
              IMPORTING
                e_value           = lwa_timepoint ).

            l_hora_limite = lwa_timepoint-time.

            TRY.
                l_zone = cl_abap_context_info=>get_user_time_zone( ).
              CATCH cx_abap_context_info_error.
            ENDTRY.

            GET TIME STAMP FIELD l_timestamp.
            CONVERT TIME STAMP l_timestamp TIME ZONE l_zone
              INTO TIME DATA(l_hora_processamento).

            IF l_hora_processamento > l_hora_limite.
              v_delivery_block_item =
                wa_brf_regras_negocio-validahorarioentradapedido.
            ENDIF.
          ENDIF.
        ENDIF.
        "-----------------------------------------------------
        " Validação de operação base
        "-----------------------------------------------------
        IF wa_brf_regras_negocio-validaoperacaobase IS NOT INITIAL.

          SELECT centro,
                 dataop AS dataoperacao,
                 horainicio,
                 horafim
            FROM zsd_ddl_i_centro_oper
            WHERE centro = @l_plant
              AND dataop <= @wa_header_sales_order_final-customerpurchaseorderdate
              ORDER BY dataop DESCENDING
            INTO @DATA(lwa_operacao_base)
            UP TO 1 ROWS.
          ENDSELECT.

          IF sy-subrc IS INITIAL.
            IF lwa_operacao_base-dataoperacao <> wa_header_sales_order_final-customerpurchaseorderdate.
              v_request_delivery_date = lwa_operacao_base-dataoperacao.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " PASSO 5: Validação de calendarização do cliente
    "-------------------------------------------------------
    IF wa_brf_regras_negocio-validacalendarizacaocliente IS NOT INITIAL.

      m_valida_calendario_cliente(
        i_v_cliente = wa_header_sales_order_final-soldtoparty
        i_v_data    = v_request_delivery_date ).

      IF v_erro_critico IS NOT INITIAL.
        RETURN.
      ENDIF.
    ENDIF.

  ENDMETHOD.

    "-------------------------------------------------------
    IF wa_header_credit_memo-creditmemorequesttype IS INITIAL.

      lwa_filtro_mensagem = VALUE #(
        bsark          = wa_header_credit_memo-creditmemorequesttype
        codigomessagem = c_constante-c_002 ).

      m_get_mensagem_brf_plus(
        EXPORTING
          i_wa_filtro_mensagem = lwa_filtro_mensagem

          e_wa_mensagem        = lwa_return_msg ).

          v_erro_critico = abap_true.
          r_v_aprovado   = abap_false.
          RETURN.
        ENDIF.
      ENDIF.

    ELSE.

      SELECT SINGLE salesdocumenttype
        FROM i_salesdocumenttype
        WHERE salesdocumenttype = @wa_header_credit_memo-creditmemorequesttype
        INTO @DATA(l_document_type).

      IF sy-subrc IS NOT INITIAL.

        lwa_filtro_mensagem = VALUE #(
          bsark          = wa_header_credit_memo-creditmemorequesttype
          codigomessagem = c_constante-c_002 ).

          EXPORTING
            i_wa_filtro_mensagem = lwa_filtro_mensagem
          IMPORTING
            e_wa_mensagem        = lwa_return_msg ).

        IF lwa_return_msg IS NOT INITIAL.
          APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                          textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.
          IF lwa_return_msg-errocritico IS NOT INITIAL.
            v_erro_critico = abap_true.
            r_v_aprovado   = abap_false.
            RETURN.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " Valida cliente
    "-------------------------------------------------------
    IF wa_header_credit_memo-soldtoparty IS INITIAL.

      lwa_filtro_mensagem = VALUE #(
        bsark          = wa_header_credit_memo-creditmemorequesttype
        codigomessagem = c_constante-c_003 ).

      m_get_mensagem_brf_plus(
        EXPORTING
          i_wa_filtro_mensagem = lwa_filtro_mensagem
        IMPORTING
          e_wa_mensagem        = lwa_return_msg ).

      IF lwa_return_msg IS NOT INITIAL.
        APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                        textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.
        IF lwa_return_msg-errocritico IS NOT INITIAL.
          v_erro_critico = abap_true.
          r_v_aprovado   = abap_false.
          RETURN.
        ENDIF.
      ENDIF.
    ENDIF.

    "-------------------------------------------------------
    " Valida itens
    "-------------------------------------------------------
    IF t_itens_credit_memo IS INITIAL.

      lwa_filtro_mensagem = VALUE #(
        bsark          = wa_header_credit_memo-creditmemorequesttype
        codigomessagem = c_constante-c_004 ).

      m_get_mensagem_brf_plus(
        EXPORTING
          i_wa_filtro_mensagem = lwa_filtro_mensagem
        IMPORTING
          e_wa_mensagem        = lwa_return_msg ).

      IF lwa_return_msg IS NOT INITIAL.
        APPEND VALUE #( codigomessagem = lwa_filtro_mensagem-codigomessagem
                        textomensagem  = lwa_return_msg-textomensagem ) TO t_log_message_error.
        IF lwa_return_msg-errocritico IS NOT INITIAL.
          v_erro_critico = abap_true.
          r_v_aprovado   = abap_false.
          RETURN.
        ENDIF.
      ENDIF.
    ENDIF.


    DATA(lt_context) = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_auart
                                                   value = wa_header_credit_memo-creditmemorequesttype )
                                                 ( name  = c_constante-c_tipooperacao
                                                   value = c_constante-c_creditmemocreate ) ).

    NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
      EXPORTING
        i_t_context       = lt_context
        i_v_function_name = c_constante-c_f_8557_dados_cabecalho
      IMPORTING
        e_value           = wa_brf_dados_cabecalho ).

    CLEAR lt_context.
    lt_context = VALUE zglcl_brf=>ty_tb_context( ( name  = c_constante-c_auart
                                                   value = wa_header_credit_memo-creditmemorequesttype )
                                                 ( name  = c_constante-c_tipooperacao
                                                   value = c_constante-c_creditmemocreate ) ).

    NEW zglcl_brf( c_constante-c_8557_microsservico )->get_value_from_brf(
      EXPORTING
        i_t_context       = lt_context
        i_v_function_name = c_constante-c_f_8557_dados_item
      IMPORTING

  ENDMETHOD.


  METHOD m_collect_rap_message.
**********************************************************************
*                                                                    *
*            ********************************************            *
*            *                IPIRANGA.                 *            *
*            *        CONFIDENCIAL E PROPRIETÁRIO       *            *
*            *       TODOS OS DIREITOS RESERVADOS       *            *
*            ********************************************            *
*                                                                    *
**********************************************************************
* NOME DO PROGRAMA    : ZSDCL_INT_ORQUESTRAD_SALES_INB               *
* TÍTULO DO PROGRAMA  : M_COLLECT_RAP_MESSAGE                        *
* DESCRIÇÃO           : Metodo coletar mensagens de erro RAP         *
* PROGRAMADOR         : EOLIVEIRA                                    *
* DATA                : 18/04/2026                                   *
* REQUEST             : DS4K902044                                   *
* RICEF               : 8557_GAP                                     *
*--------------------------------------------------------------------*
* HISTÓRICO DE MUDANÇAS:                                             *
* MUD  DATA        AUTOR          REQUEST #    DESCRIÇÃO             *
* 001  18/04/2026  EOLIVEIRA      DS4K902044   CÓDIGO INICIAL        *
**********************************************************************
    CHECK i_o_message IS BOUND.

    TRY.
        DATA(lo_t100) = CAST if_t100_dyn_msg( i_o_message ).
        DATA(l_msgty) = lo_t100->msgty.
      CATCH cx_sy_move_cast_error.
    ENDTRY.

    IF l_msgty <> c_constante-c_e.
      RETURN.
    ENDIF.

    DATA(lo_t100_msg) = CAST if_t100_message( i_o_message ).

    APPEND VALUE #(
      msgid    = lo_t100_msg->t100key-msgid
      msgno    = lo_t100_msg->t100key-msgno
      msgv1    = lo_t100->msgv1
      msgv2    = lo_t100->msgv2
      msgv3    = lo_t100->msgv3
      msgv4    = lo_t100->msgv4
      severity = i_o_message->m_severity
    ) TO t_rap_messages.

  ENDMETHOD.

  METHOD m_mapping_bo_creditmemo_create.
**********************************************************************
*                                                                    *
*            ********************************************            *
*            *                IPIRANGA.                 *            *
*            *        CONFIDENCIAL E PROPRIETÁRIO       *            *
*            *       TODOS OS DIREITOS RESERVADOS       *            *
*            ********************************************            *
*                                                                    *
**********************************************************************
* NOME DO PROGRAMA    : ZSDCL_INT_ORQUESTRAD_SALES_INB               *
* TÍTULO DO PROGRAMA  : M_MAPPING_BO_CREDITMEMO_CREATE               *
* DESCRIÇÃO           : Metodo mapeamento BO criação do Credit Memo  *
* PROGRAMADOR         : EOLIVEIRA                                    *
* DATA                : 18/04/2026                                   *
* REQUEST             : DS4K902044                                   *
* RICEF               : 8557_GAP                                     *
*--------------------------------------------------------------------*
* HISTÓRICO DE MUDANÇAS:                                             *
* MUD  DATA        AUTOR          REQUEST #    DESCRIÇÃO             *
* 001  18/04/2026  EOLIVEIRA      DS4K902044   CÓDIGO INICIAL        *
**********************************************************************
    DATA: lwa_mapped   TYPE RESPONSE FOR MAPPED   i_creditmemorequesttp,
          lwa_failed   TYPE RESPONSE FOR FAILED   i_creditmemorequesttp,
          lwa_reported TYPE RESPONSE FOR REPORTED i_creditmemorequesttp.

    MODIFY ENTITIES OF i_creditmemorequesttp

      "-------------------------------------------------------
      " HEADER
      "-------------------------------------------------------
      ENTITY creditmemorequest
        CREATE AUTO FILL CID
        FIELDS ( creditmemorequesttype
                 salesorganization
                 distributionchannel
                 organizationdivision
                 soldtoparty
                 purchaseorderbycustomer
                 sddocumentreason )

        WITH VALUE #(
          ( %cid  = c_constante-c_h001
            %data = VALUE #(
              creditmemorequesttype   = wa_header_credit_memo-creditmemorequesttype
              salesorganization       = wa_brf_dados_cabecalho-vkorg
              distributionchannel     = wa_brf_dados_cabecalho-vtweg
              organizationdivision    = wa_brf_dados_cabecalho-spart
              soldtoparty             = wa_header_credit_memo-soldtoparty
              purchaseorderbycustomer = wa_header_credit_memo-purchaseorderbycustomer
              sddocumentreason        = wa_header_credit_memo-sddocumentreason ) ) )

      "-------------------------------------------------------
      " ITEMS
      "-------------------------------------------------------
      CREATE BY \_item AUTO FILL CID
        FIELDS ( product
                 plant )

        WITH VALUE #(
          FOR lwa_item IN t_itens_credit_memo
          ( %cid_ref = c_constante-c_h001
            %target  = VALUE #(
              ( %cid    = lwa_item-cid
                product = lwa_item-material
                plant   = lwa_item-plant ) ) ) )

      "-------------------------------------------------------
      " ITEM PRICING
      "-------------------------------------------------------
      ENTITY creditmemorequestitem
        CREATE BY \_itempricingelement AUTO FILL CID
        FIELDS ( conditiontype
                 conditionrateamount )

        WITH VALUE #(
          FOR lwa_item  IN t_itens_credit_memo
          FOR lwa_price IN t_pricing_credit_memo
            WHERE ( cid_ref = lwa_item-cid )
          ( %cid_ref = lwa_item-cid
            %target  = VALUE #(
              ( %cid                = lwa_price-cid
                conditiontype       = wa_brf_dados_item-kschl "'ZPMN'
                conditionrateamount = CONV #( lwa_price-conditionratevalue ) ) ) ) )

      MAPPED   lwa_mapped
      FAILED   lwa_failed
      REPORTED lwa_reported.

    wa_mapped_credit_memo = lwa_mapped.

    IF lwa_failed IS NOT INITIAL OR lwa_reported IS NOT INITIAL .

      v_erro_critico = abap_true.

      "-----------------------------------------------------
      " Header
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemorequest INTO DATA(lwa_rep_hd).
        CHECK lwa_rep_hd-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_hd-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Item
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemorequestitem INTO DATA(lwa_rep_item).
        CHECK lwa_rep_item-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Item Pricing Element
      "-----------------------------------------------------
      LOOP AT lwa_reported-crdtmemoreqitempricingelement INTO DATA(lwa_rep_item_price).
        CHECK lwa_rep_item_price-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item_price-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Header Pricing Element
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemoreqpricingelement INTO DATA(lwa_rep_hd_price).
        CHECK lwa_rep_hd_price-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_hd_price-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Item Partner
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemorequestitempartner INTO DATA(lwa_rep_item_partner).
        CHECK lwa_rep_item_partner-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item_partner-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Item Text
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemorequestitemtext INTO DATA(lwa_rep_item_text).
        CHECK lwa_rep_item_text-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_item_text-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Header Partner
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemorequestpartner INTO DATA(lwa_rep_partner).
        CHECK lwa_rep_partner-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_partner-%msg ).
      ENDLOOP.

      "-----------------------------------------------------
      " Header Text
      "-----------------------------------------------------
      LOOP AT lwa_reported-creditmemorequesttext INTO DATA(lwa_rep_text).
        CHECK lwa_rep_text-%msg IS BOUND.
        m_collect_rap_message( lwa_rep_text-%msg ).
      ENDLOOP.

    ENDIF.

  ENDMETHOD.
